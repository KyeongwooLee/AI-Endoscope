\[ \text{Daily Calorie Deficit} = \text{Calories Burned} - \text{Calories Eaten} \]

\[ \text{Total Calorie Deficit for 10 Pounds} = \text{Daily Calorie Deficit} \times 10 \]

\[ \text{Number of Days} = \frac{\text{Total Calorie Deficit for 10 Pounds}}{\text{Daily Calorie Deficit}} \]
